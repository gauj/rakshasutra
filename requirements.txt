# --- Environment variables (all optional) ---
# BEE_NOTES=1          start the Bee bridge (stream the guardian's utterances, write reminders)
# BEE_CLI_BINARY=...   full path to the bee executable
# BEE_RAW_LOG=1        log raw Bee stream events
# BEDROCK_MODEL_ID=... turn on Amazon Bedrock for reports and Q&A (needs AWS credentials)
# AWS_REGION=...       Bedrock region, default us-east-1
# BEE_ALERTS=1         also send critical alerts to the guardian's Bee as reminders (writes to the real Bee account)
# SNS_TO_PHONE=...     text critical alerts to the parent's own phone through Amazon SNS (needs @aws-sdk/client-sns and AWS credentials; emergency numbers are refused)
# PARENT_TOKEN=...     demo parent credential, default demo-parent
# PORT=...             port, default 8787
# HOST=...             listen address, default 127.0.0.1 (this computer only)
# DATA_DIR=...         where the private family data lives, default data/
# NTFY_TOPIC=...       push alerts through ntfy.sh (no location, no name in the message)
# FORCE_DEMO=1         run the demo even if a private data/ folder exists

# --- Commands ---
# npm test             run the 75 unit tests
# npm run setup:real   create the private family data and the family code
# npm run start:real   build the app and start it in real mode
# npm run sim:serve    private simulator server on :8788;  npm run sim -- <scenario>
# npm run e2e          37 demo API checks (needs a demo-mode server on :8787)
# npm run e2e:real     28 real-mode checks (starts its own server)
# npm run typecheck    typecheck all three packages
# npm run dev:server   API on http://localhost:8787
# npm run dev:web      parent app on http://localhost:5173
